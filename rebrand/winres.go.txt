// Ghostline Windows resource generator for the Tailscale fork (copied into the Tailscale tree as
// cmd/ghostline-winres/main.go by build.sh and removed again after the run).
//
// It writes the manifest_windows_<arch>.syso files that cmd/tailscaled and cmd/tailscale link,
// with the same RT_MANIFEST (ID 1) that cmd/mkmanifest puts there, plus the application icon and a
// VERSIONINFO block, so Explorer and the file properties show your product and company names.
// It uses github.com/tc-hib/winres, which the Tailscale tree already requires (go.mod, used by
// cmd/mkmanifest), so it needs no extra downloads beyond the tree's own modules.
//
// Usage (from the Tailscale source root):
//
//	go run ./cmd/ghostline-winres -manifest cmd/tailscaled/windows-manifest.xml -ico ghostline.ico \
//	  -product "Ghostline" -company "..." -desc "Ghostline service" -exe ghostlined.exe \
//	  -version 1.104.5 -out cmd/tailscaled
package main

import (
	"flag"
	"fmt"
	"log"
	"os"
	"path/filepath"
	"strings"

	"github.com/tc-hib/winres"
	"github.com/tc-hib/winres/version"
)

func main() {
	manifest := flag.String("manifest", "", "application manifest XML (windows-manifest.xml)")
	ico := flag.String("ico", "", "multi-resolution .ico file")
	product := flag.String("product", "", "ProductName")
	company := flag.String("company", "", "CompanyName")
	desc := flag.String("desc", "", "FileDescription")
	exe := flag.String("exe", "", "OriginalFilename, e.g. ghostlined.exe")
	ver := flag.String("version", "", "file/product version, e.g. 1.104.5")
	copyright := flag.String("copyright", "", "LegalCopyright")
	out := flag.String("out", "", "output directory (gets manifest_windows_{amd64,386,arm64}.syso)")
	flag.Parse()
	if *manifest == "" || *ico == "" || *product == "" || *exe == "" || *ver == "" || *out == "" {
		flag.Usage()
		os.Exit(2)
	}

	manifestData, err := os.ReadFile(*manifest)
	if err != nil {
		log.Fatalf("reading manifest: %v", err)
	}
	f, err := os.Open(*ico)
	if err != nil {
		log.Fatalf("opening icon: %v", err)
	}
	icon, err := winres.LoadICO(f)
	f.Close()
	if err != nil {
		log.Fatalf("loading icon %s: %v", *ico, err)
	}

	rs := winres.ResourceSet{}
	// Same manifest resource as cmd/mkmanifest.
	if err := rs.Set(winres.RT_MANIFEST, winres.ID(1), 0, manifestData); err != nil {
		log.Fatalf("manifest: %v", err)
	}
	// Named icon groups sort before numeric ones, so Explorer uses this one for the .exe.
	if err := rs.SetIcon(winres.Name("APPICON"), icon); err != nil {
		log.Fatalf("icon: %v", err)
	}

	v := *ver
	if strings.Count(v, ".") == 2 {
		v += ".0"
	}
	vi := version.Info{}
	vi.SetFileVersion(v)
	vi.SetProductVersion(v)
	const lang = 0x0409 // en-US, the language Explorer falls back to
	strs := map[string]string{
		"ProductName":      *product,
		"CompanyName":      *company,
		"FileDescription":  *desc,
		"OriginalFilename": *exe,
		"InternalName":     strings.TrimSuffix(*exe, ".exe"),
		"LegalCopyright":   *copyright,
		"FileVersion":      v,
		"ProductVersion":   v,
	}
	for k, s := range strs {
		if s == "" {
			continue
		}
		if err := vi.Set(lang, k, s); err != nil {
			log.Fatalf("version string %s: %v", k, err)
		}
	}
	rs.SetVersionInfo(vi)

	for _, arch := range []winres.Arch{winres.ArchAMD64, winres.ArchI386, winres.ArchARM64} {
		name := filepath.Join(*out, fmt.Sprintf("manifest_windows_%s.syso", arch))
		w, err := os.Create(name)
		if err != nil {
			log.Fatalf("creating %s: %v", name, err)
		}
		if err := rs.WriteObject(w, arch); err != nil {
			log.Fatalf("writing %s: %v", name, err)
		}
		if err := w.Close(); err != nil {
			log.Fatalf("closing %s: %v", name, err)
		}
		fmt.Println("wrote", name)
	}
}
